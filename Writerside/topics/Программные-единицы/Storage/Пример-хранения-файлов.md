# Пример хранения файлов

```Python
from slinn import Storage


if __name__ == '__main__':
    #
    # 1. Хранилище из каталога
    #
    storage = Storage(root='./static')
    
    # Проверка, существует ли файл
    if storage.isfile('style.css'):
        # Открытие файла
        with storage('style.css', 'r') as f:
            css_content = f.read()
            print(f'{len(css_content.split('\n'))} cтрок в CSS-файле')
    
    # Получаем список файлов в каталоге
    files = storage.listdir('/')
    print('Файлы в каталоге /static:', files)
    
    #
    # 2. Хранилище из `.zip`-архива
    #
    zip_storage = Storage(zip_file='./data.zip')
    
    # Проверка, является ли путь каталогом
    if zip_storage.isdir('images'):
        # Список файлов внутри архива
        images = zip_storage.listdir('images')
        print('Картинки в каталоге /images:', images)
    
        # Открытие файла
        with zip_storage('images/logo.png', 'rb') as f:
            logo_data = f.read()
            print(f'Размер лого {len(logo_data)}Б')
```
